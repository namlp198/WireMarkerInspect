cmake_minimum_required(VERSION 3.20)
project(WireMarkerVision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs features2d calib3d)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime SDK root")
if(NOT EXISTS "${ONNXRUNTIME_ROOT}/include/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to the installed SDK.")
endif()
add_library(WireMarkerVision SHARED vision.cpp terminal_matching.cpp)
target_include_directories(WireMarkerVision PRIVATE ${OpenCV_INCLUDE_DIRS} "${ONNXRUNTIME_ROOT}/include")
target_link_libraries(WireMarkerVision PRIVATE ${OpenCV_LIBS} "${ONNXRUNTIME_ROOT}/lib/onnxruntime.lib")
target_compile_definitions(WireMarkerVision PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(WireMarkerVision PRIVATE /W4 /EHsc /utf-8)
add_custom_command(TARGET WireMarkerVision POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:WireMarkerVision>")
enable_testing()
add_executable(VisionNativeTests tests.cpp)
target_link_libraries(VisionNativeTests PRIVATE WireMarkerVision ${OpenCV_LIBS})
target_include_directories(VisionNativeTests PRIVATE ${OpenCV_INCLUDE_DIRS})
add_test(NAME native_contract COMMAND VisionNativeTests)
add_executable(VisionOcrCli ocr_cli.cpp)
target_link_libraries(VisionOcrCli PRIVATE WireMarkerVision ${OpenCV_LIBS})
target_include_directories(VisionOcrCli PRIVATE ${OpenCV_INCLUDE_DIRS})
